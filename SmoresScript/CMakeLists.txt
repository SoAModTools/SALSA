cmake_minimum_required(VERSION 3.20)
project(SmoresScript LANGUAGES CXX)
set(SPICE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../third-party/SPICE" CACHE PATH "SPICE include root")
add_library(SmoresScript STATIC src/Relationships.cpp)
target_compile_features(SmoresScript PUBLIC cxx_std_20)
target_include_directories(SmoresScript PUBLIC include "${SPICE_ROOT}")
add_executable(SmoresScriptTests tests/RelationshipsTests.cpp)
target_link_libraries(SmoresScriptTests PRIVATE SmoresScript)
enable_testing()
add_test(NAME SmoresScriptContracts COMMAND SmoresScriptTests)
